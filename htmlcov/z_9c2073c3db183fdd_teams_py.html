119def _norm(s: str) -> str: 

120 return re.sub(r"[^a-z0-9]", "", s.lower())

123def _build_aliases() -> dict[str, str]: 

124 aliases: dict[str, str] = {} 

125 for t in TEAMS.values(): 

126 city_nick = t.name.split(" ") 

127 aliases[_norm(t.abbr)] = t.abbr 

128 aliases[_norm(t.name)] = t.abbr 

129 aliases[_norm(city_nick[-1])] = t.abbr # nickname, e.g. "celtics" 

130 aliases[_norm(" ".join(city_nick[:-1]))] = t.abbr # city, e.g. "boston" 

131 # "Trail Blazers" has a two-word nickname; "Los Angeles" is ambiguous. 

132 aliases.pop(_norm("Portland Trail"), None) 

133 aliases[_norm("Portland")] = "POR" 

134 aliases.pop(_norm("Los Angeles"), None) 

135 aliases.pop(_norm("LA"), None) 

136 aliases.update(_EXTRA_ALIASES) 

137 return aliases

140ALIASES = _build_aliases()

143class UnknownTeamError(ValueError): 

144 pass

147def resolve(name: str) -> str: 

148 """Return the canonical abbreviation for any known spelling.""" 

149 key = _norm(name) 

150 if key in ALIASES: 

151 return ALIASES[key] 

152 raise UnknownTeamError(f"unknown NBA team: {name!r}") 

153 

154 

155def team(abbr_or_name: str) -> Team: 

156 return TEAMS[resolve(abbr_or_name)]

159def travel_km(from_abbr: str, to_abbr: str) -> float: 

160 """Great-circle distance between two home arenas in kilometres.""" 

161 a, b = team(from_abbr), team(to_abbr) 

162 lat1, lon1, lat2, lon2 = map(math.radians, (a.lat, a.lon, b.lat, b.lon)) 

163 h = ( 

164 math.sin((lat2 - lat1) / 2) ** 2 

165 + math.cos(lat1) * math.cos(lat2) * math.sin((lon2 - lon1) / 2) ** 2 

166 ) 

167 return 2 * 6371.0 * math.asin(math.sqrt(h))
