1"""NBA box scores and schedule via nba_api, cached locally and stored as-of. 

2 

3Availability rules (see ``timeutil.box_score_known_at``): 

4 

5* A game's final score and every player's minutes and plus-minus become 

6 known at 08:00 UTC the day after the game, safely after any final. 

7* A scheduled game is known from the earlier of when we fetched it and 24h 

8 before tip; NBA schedules are published months ahead. 

9 

10stats.nba.com blocks many datacenter IPs, so run this from a home connection. 

11Raw responses are cached under ``cache/nba_api`` (gitignored). Completed 

12seasons are cached permanently; the current season is re-fetched on request. 

13"""

33# GAME_ID prefixes: 002 regular season, 004 playoffs, 005 play-in. Preseason

35COMPETITIVE_PREFIXES = ("002", "004", "005")

38def nba_api_fetcher(endpoint: str, params: dict[str, Any], pause_s: float = 1.5) -> pd.DataFrame: 

39 """Call nba_api politely: one request at a time with a pause in between.""" 

40 from nba_api.stats.endpoints import leaguegamelog, scheduleleaguev2 

41 

42 time.sleep(pause_s) 

43 if endpoint == "leaguegamelog": 

44 return leaguegamelog.LeagueGameLog(timeout=60, **params).get_data_frames()[0] 

45 if endpoint == "schedule": 

46 return scheduleleaguev2.ScheduleLeagueV2(timeout=60, **params).get_data_frames()[0] 

47 raise ValueError(f"unknown endpoint: {endpoint}")

50def parse_minutes(value: Any) -> float: 

51 """nba_api reports minutes as a number or as 'MM:SS'.""" 

52 if value is None or (isinstance(value, float) and pd.isna(value)): 

53 return 0.0 

54 if isinstance(value, str) and ":" in value: 

55 mm, ss = value.split(":", 1) 

56 return int(mm) + int(ss) / 60 

57 return float(value)

60def _read_cache(path: Path) -> pd.DataFrame: 

61 try: 

62 return pd.read_csv(path, dtype={"GAME_ID": str, "gameId": str}) 

63 except pd.errors.EmptyDataError: # e.g. playoffs not played yet 

64 return pd.DataFrame()

67class NbaStats: 

68 def __init__(self, cache_dir: str | Path = "cache/nba_api", fetcher: Fetcher | None = None): 

69 self.cache_dir = Path(cache_dir) 

70 self.fetcher = fetcher or nba_api_fetcher 

71 

72 def _cached(self, endpoint: str, params: dict[str, Any], refresh: bool) -> pd.DataFrame: 

73 key = "_".join([endpoint, *(f"{k}-{v}" for k, v in sorted(params.items()))]) 

74 path = self.cache_dir / f"{key.replace(' ', '')}.csv" 

75 if path.exists() and not refresh: 

76 return _read_cache(path) 

77 df = self.fetcher(endpoint, params) 

78 self.cache_dir.mkdir(parents=True, exist_ok=True) 

79 df.to_csv(path, index=False) 

80 return _read_cache(path)

99def _competitive(game_id: str) -> bool: 

100 return str(game_id).zfill(10)[:3] in COMPETITIVE_PREFIXES

103def games_from_team_logs(team_logs: pd.DataFrame, season: str) -> list[dict[str, Any]]: 

104 """Pair the two team rows of each game into one home/away record.""" 

105 out = [] 

106 for game_id, grp in team_logs.groupby("GAME_ID"): 

107 game_id = str(game_id).zfill(10) 

108 if not _competitive(game_id): 

109 continue 

110 if len(grp) != 2: 110 ↛ 111line 110 didn't jump to line 111 because the condition on line 110 was never true

111 log.warning("game %s has %d team rows; skipped", game_id, len(grp)) 

112 continue 

113 home_rows = grp[grp["MATCHUP"].str.contains(" vs. ", regex=False)] 

114 away_rows = grp[grp["MATCHUP"].str.contains(" @ ", regex=False)] 

115 if len(home_rows) != 1 or len(away_rows) != 1: 115 ↛ 116line 115 didn't jump to line 116 because the condition on line 115 was never true

116 log.warning("game %s has no clear home/away; skipped", game_id) 

117 continue 

118 h, a = home_rows.iloc[0], away_rows.iloc[0] 

119 game_day = date.fromisoformat(str(h["GAME_DATE"])[:10]) 

120 out.append( 

121 { 

122 "game_id": game_id, 

123 "season": season, 

124 "game_date": game_day.isoformat(), 

125 "home": resolve(h["TEAM_ABBREVIATION"]), 

126 "away": resolve(a["TEAM_ABBREVIATION"]), 

127 "home_pts": int(h["PTS"]), 

128 "away_pts": int(a["PTS"]), 

129 "known_at": box_score_known_at(game_day), 

130 } 

131 ) 

132 return out

135def possessions(fga: float, fta: float, oreb: float, tov: float) -> float: 

136 """Standard single-team possession estimate.""" 

137 return fga + 0.44 * fta - oreb + tov

140def team_rows(team_logs: pd.DataFrame, games: list[dict[str, Any]]) -> list[dict[str, Any]]: 

141 """Per-team box totals for games already paired by ``games_from_team_logs``.""" 

142 by_id = {g["game_id"]: g for g in games} 

143 out = [] 

144 for rec in team_logs.to_dict("records"): 

145 game_id = str(rec["GAME_ID"]).zfill(10) 

146 g = by_id.get(game_id) 

147 if g is None: 

148 continue 

149 team = resolve(rec["TEAM_ABBREVIATION"]) 

150 out.append( 

151 { 

152 "game_id": game_id, 

153 "team": team, 

154 "opp": g["away"] if team == g["home"] else g["home"], 

155 "is_home": team == g["home"], 

156 "pts": int(rec["PTS"]), 

157 "fga": int(rec["FGA"]), 

158 "fta": int(rec["FTA"]), 

159 "oreb": int(rec["OREB"]), 

160 "tov": int(rec["TOV"]), 

161 "poss": possessions(rec["FGA"], rec["FTA"], rec["OREB"], rec["TOV"]), 

162 "known_at": g["known_at"], 

163 } 

164 ) 

165 return out

168def player_rows(player_logs: pd.DataFrame) -> list[dict[str, Any]]: 

169 out = [] 

170 for rec in player_logs.to_dict("records"): 

171 game_id = str(rec["GAME_ID"]).zfill(10) 

172 if not _competitive(game_id): 

173 continue 

174 game_day = date.fromisoformat(str(rec["GAME_DATE"])[:10]) 

175 pm = rec.get("PLUS_MINUS") 

176 out.append( 

177 { 

178 "game_id": game_id, 

179 "player_id": int(rec["PLAYER_ID"]), 

180 "player_name": str(rec["PLAYER_NAME"]), 

181 "team": resolve(rec["TEAM_ABBREVIATION"]), 

182 "minutes": parse_minutes(rec.get("MIN")), 

183 "plus_minus": None if pm is None or pd.isna(pm) else float(pm), 

184 "pts": None if pd.isna(rec.get("PTS")) else int(rec["PTS"]), 

185 "started": None, 

186 "known_at": box_score_known_at(game_day), 

187 } 

188 ) 

189 return out

192def ingest_season( 

193 store, 

194 stats: NbaStats, 

195 season: str, 

196 *, 

197 season_types: tuple[str, ...] = ("Regular Season", "PlayIn", "Playoffs"), 

198 refresh: bool = False, 

199) -> dict[str, int]: 

200 """Load one season's results and player lines into the store.""" 

201 counts = {"games": 0, "player_games": 0} 

202 for st in season_types: 

203 teams = stats.game_logs(season, "T", st, refresh) 

204 if teams.empty: 

205 continue 

206 games = games_from_team_logs(teams, season) 

207 for g in games: 

208 store.upsert_game( 

209 game_id=g["game_id"], 

210 season=season, 

211 game_date=g["game_date"], 

212 home=g["home"], 

213 away=g["away"], 

214 known_at=g["known_at"] - timedelta(days=60), 

215 ) 

216 store.upsert_result( 

217 game_id=g["game_id"], 

218 home_pts=g["home_pts"], 

219 away_pts=g["away_pts"], 

220 known_at=g["known_at"], 

221 ) 

222 store.upsert_team_games(team_rows(teams, games)) 

223 known_ids = {g["game_id"] for g in games} 

224 players = [ 

225 r 

226 for r in player_rows(stats.game_logs(season, "P", st, refresh)) 

227 if r["game_id"] in known_ids 

228 ] 

229 store.upsert_player_games(players) 

230 counts["games"] += len(games) 

231 counts["player_games"] += len(players) 

232 return counts

235def schedule_rows(schedule: pd.DataFrame, season: str, now: datetime) -> list[dict[str, Any]]: 

236 out = [] 

237 for rec in schedule.to_dict("records"): 

238 game_id = str(rec["gameId"]).zfill(10) 

239 if not _competitive(game_id): 239 ↛ 240line 239 didn't jump to line 240 because the condition on line 239 was never true

240 continue 

241 try: 

242 tip = parse_iso(str(rec["gameDateTimeUTC"])) 

243 home = resolve(str(rec["homeTeam_teamTricode"])) 

244 away = resolve(str(rec["awayTeam_teamTricode"])) 

245 except ValueError: 

246 continue # TBD playoff slots have no teams or time yet 

247 out.append( 

248 { 

249 "game_id": game_id, 

250 "season": season, 

251 "game_date": str(rec["gameDateEst"])[:10], 

252 "home": home, 

253 "away": away, 

254 "tip_utc": tip, 

255 "known_at": min(now, tip - timedelta(days=1)), 

256 } 

257 ) 

258 return out
